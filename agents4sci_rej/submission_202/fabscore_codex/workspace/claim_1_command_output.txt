   237	    imgs = torch.stack([item[0] for item in batch])  # (B,3,28,28)
   238	    # We'll merge the three images (for a claim) along channel dim: (B, 3*3, 28, 28)
   239	    imgs = imgs.view(len(batch), -1, 28, 28)  # (B,9,28,28)
   240	    input_ids = torch.stack([item[1] for item in batch])
   241	    attn_mask = torch.stack([item[2] for item in batch])
   242	    labels = torch.stack([item[3] for item in batch])
   243	    logicvec = torch.stack([item[4] for item in batch])
   244	    return imgs, input_ids, attn_mask, labels, logicvec

   247	# Adjusted CNN Vision Encoder for merged image representation
   248	class CNNVisionEncoder(nn.Module):
   249	    def __init__(self, input_channels=9):
   250	        super().__init__()
   251	        self.net = nn.Sequential(
   252	            nn.Conv2d(input_channels, 32, 3, padding=1),  # 32x28x28
   253	            nn.ReLU(),
   254	            nn.Conv2d(32, 64, 3, padding=1),  # 64x28x28
   255	            nn.ReLU(),
   256	            nn.MaxPool2d(2),  # 64x14x14
   257	            nn.Conv2d(64, 128, 3, padding=1),  # 128x14x14
   258	            nn.ReLU(),
   259	            nn.MaxPool2d(2),  # 128x7x7
   260	            nn.Flatten(),  # 128*7*7
   261	            nn.Linear(128 * 7 * 7, 256),
   262	            nn.ReLU(),
   263	        )
   264	
   265	    def forward(self, x):
   266	        return self.net(x)
   267	
   268	
   269	class ClaimVerifier(nn.Module):
   270	    def __init__(self, in_c, do_finetune_text=True):
   271	        super().__init__()
   272	        self.vision = CNNVisionEncoder(input_channels=in_c)
   273	        self.text = BertModel.from_pretrained("bert-base-uncased")
   274	        for param in self.text.parameters():
   275	            param.requires_grad = do_finetune_text
   276	        self.fc = nn.Sequential(
   277	            nn.Linear(256 + 768, 192),
   278	            nn.ReLU(),
   279	            nn.Linear(192, 1),
   280	            nn.Sigmoid(),
   281	        )
   282	
   283	    def forward(self, imgs, input_ids, attn_mask):
   284	        vis_feat = self.vision(imgs)
   285	        txt_feat = self.text(
   286	            input_ids=input_ids, attention_mask=attn_mask
   287	        ).last_hidden_state[:, 0, :]
   288	        x = torch.cat([vis_feat, txt_feat], dim=1)
   289	        return self.fc(x).squeeze(1)

   292	def logical_consistency_accuracy(preds, gts, logicvecs, claim_labels=None):
   293	    # For this research, logical consistency is defined as (rounded) prediction equals ground truth
   294	    correct = np.round(preds) == gts
   295	    return np.mean(correct)

   298	def train_on_dataset(name, dataset, in_c):
   299	    print(f"\nTraining on {name} ...")
   300	    train_len = int(0.85 * len(dataset))
   301	    val_len = len(dataset) - train_len
   302	    train_set, val_set = random_split(
   303	        dataset, [train_len, val_len], generator=torch.Generator().manual_seed(42)
   304	    )
   305	    train_loader = DataLoader(
   306	        train_set,
   307	        batch_size=BATCH_SIZE,
   308	        shuffle=True,
   309	        collate_fn=collate_fn,
   310	        num_workers=2,
   311	        pin_memory=True,
   312	    )
   313	    val_loader = DataLoader(
   314	        val_set,
   315	        batch_size=BATCH_SIZE,
   316	        shuffle=False,
   317	        collate_fn=collate_fn,
   318	        num_workers=2,
   319	        pin_memory=True,
   320	    )
   321	    model = ClaimVerifier(in_c, do_finetune_text=True).to(device)
   322	    optimizer = optim.Adam(model.parameters(), lr=LR)
   323	    criterion = nn.BCELoss()
   324	
   325	    for epoch in range(NUM_EPOCHS):
   326	        model.train()
   327	        total_loss = 0.0
   328	        all_preds, all_gts, all_logicvecs = [], [], []
   329	        correct = 0
   330	        for imgs, input_ids, attn_mask, labels, logicvec in train_loader:
   331	            imgs = imgs.to(device).float()
   332	            input_ids = input_ids.to(device)
   333	            attn_mask = attn_mask.to(device)
   334	            labels = labels.to(device)
   335	            output = model(imgs, input_ids, attn_mask)
   336	            loss = criterion(output, labels)
   337	            optimizer.zero_grad()
   338	            loss.backward()
   339	            optimizer.step()
   340	            total_loss += loss.item() * len(labels)
   341	            preds = (output > 0.5).float()
   342	            correct += (preds == labels).sum().item()
   343	            all_preds.append(preds.detach().cpu().numpy())
   344	            all_gts.append(labels.detach().cpu().numpy())
   345	            all_logicvecs.append(logicvec.cpu().numpy())
   346	        ntrain = train_len
   347	        train_acc = correct / ntrain
   348	        train_loss = total_loss / ntrain
   349	        tpreds = np.concatenate(all_preds)
   350	        tgts = np.concatenate(all_gts)
   351	        tlogicvecs = np.concatenate(all_logicvecs)
   352	        train_logic = logical_consistency_accuracy(tpreds, tgts, tlogicvecs)
   353	        experiment_data[name]["metrics"]["train"].append(train_acc)
   354	        experiment_data[name]["losses"]["train"].append(train_loss)
   355	        experiment_data[name]["metrics"]["train_logic"].append(train_logic)
   356	        # Validation
   357	        model.eval()
   358	        val_loss = 0.0
   359	        all_preds, all_gts, all_logicvecs = [], [], []
   360	        vcorrect = 0
   361	        with torch.no_grad():
   362	            for imgs, input_ids, attn_mask, labels, logicvec in val_loader:
   363	                imgs = imgs.to(device).float()
   364	                input_ids = input_ids.to(device)
   365	                attn_mask = attn_mask.to(device)
   366	                labels = labels.to(device)
   367	                output = model(imgs, input_ids, attn_mask)
   368	                loss = criterion(output, labels)
   369	                val_loss += loss.item() * len(labels)
   370	                preds = (output > 0.5).float()
   371	                vcorrect += (preds == labels).sum().item()
   372	                all_preds.append(preds.detach().cpu().numpy())
   373	                all_gts.append(labels.detach().cpu().numpy())
   374	                all_logicvecs.append(logicvec.cpu().numpy())
   375	        nval = val_len
   376	        val_acc = vcorrect / nval
   377	        val_loss = val_loss / nval
   378	        vpreds = np.concatenate(all_preds)
   379	        vgts = np.concatenate(all_gts)
   380	        vlogicvecs = np.concatenate(all_logicvecs)
   381	        val_logic = logical_consistency_accuracy(vpreds, vgts, vlogicvecs)
   382	        experiment_data[name]["metrics"]["val"].append(val_acc)
   383	        experiment_data[name]["losses"]["val"].append(val_loss)
   384	        experiment_data[name]["metrics"]["val_logic"].append(val_logic)
   385	        experiment_data[name]["epochs"].append(epoch + 1)
   386	        print(
   387	            f"Epoch {epoch+1}: validation_loss = {val_loss:.4f}  | val_acc = {val_acc:.4f}  | val_logic_acc = {val_logic:.4f}"
   388	        )
   389	        if epoch == NUM_EPOCHS - 1:
   390	            experiment_data[name]["predictions"] = vpreds
   391	            experiment_data[name]["ground_truth"] = vgts
