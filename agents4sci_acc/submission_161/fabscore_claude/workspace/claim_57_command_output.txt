### Command 1
python3 -c "
import pandas as pd

df = pd.read_csv('agent4science/metadata/endocrine_datasets_organized_table.csv')

# Split Tissue_Category by '; ' and get all unique values
all_categories = set()
for val in df['Tissue_Category'].dropna():
    for cat in val.split('; '):
        all_categories.add(cat.strip())

print(f'Total distinct tissue categories: {len(all_categories)}')
print(f'Categories: {sorted(all_categories)}')
print()
print(f'Total rows in CSV: {len(df)}')
"
