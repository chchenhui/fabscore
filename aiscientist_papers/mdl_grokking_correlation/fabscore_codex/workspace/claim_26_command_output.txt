import numpy as np
paths = {
  'x_div_y': 'fabscore_codex/workspace/claim_25_x_div_y/all_results.npy',
  'x_minus_y': 'fabscore_codex/workspace/claim_26_cross_dataset/x_minus_y/all_results.npy',
  'x_plus_y': 'fabscore_codex/workspace/claim_26_cross_dataset/x_plus_y/all_results.npy',
}
for ds, path in paths.items():
    results = np.load(path, allow_pickle=True).item()
    steps = np.array([x['step'] for x in results[f'{ds}_0_val_info']])
    mean_val = np.mean([[x['val_accuracy'] for x in results[f'{ds}_{i}_val_info']] for i in range(3)], axis=0)
    mdl_steps = np.array([x['step'] for x in results[f'{ds}_0_mdl_info']])
    mdl = np.array([x['mdl'] for x in results[f'{ds}_0_mdl_info']])
    mdltp = int(mdl_steps[int(np.argmin(np.diff(mdl)))])
    grok = next((int(s) for s, a in zip(steps, mean_val) if a >= 0.95), None)
    print(ds, {'mdl_transition_point': mdltp, 'grokking_point_95': grok})
