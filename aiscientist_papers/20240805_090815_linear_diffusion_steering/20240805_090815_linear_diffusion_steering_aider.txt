class NoiseScheduler():
    def __init__(
            self,
            num_timesteps=1000,
            beta_start=0.0001,
            beta_end=0.02,
            beta_schedule="linear",
    ):
        self.num_timesteps = num_timesteps
        if beta_schedule == "linear":
            self.betas = torch.linspace(
                beta_start, beta_end, num_timesteps, dtype=torch.float32).to(device)
        elif beta_schedule == "quadratic":
            self.betas = (torch.linspace(
                beta_start ** 0.5, beta_end ** 0.5, num_timesteps, dtype=torch.float32) ** 2).to(device)
        else:
            raise ValueError(f"Unknown beta schedule: {beta_schedule}")

        self.alphas = 1.0 - self.betas
        self.alphas_cumprod = torch.cumprod(self.alphas, axis=0).to(device)
        self.alphas_cumprod_prev = F.pad(self.alphas_cumprod[:-1], (1, 0), value=1.).to(device)

        # required for self.add_noise
        self.sqrt_alphas_cumprod = (self.alphas_cumprod ** 0.5).to(device)
        self.sqrt_one_minus_alphas_cumprod = ((1 - self.alphas_cumprod) ** 0.5).to(device)

        # required for reconstruct_x0
        self.sqrt_inv_alphas_cumprod = torch.sqrt(1 / self.alphas_cumprod).to(device)
        self.sqrt_inv_alphas_cumprod_minus_one = torch.sqrt(
            1 / self.alphas_cumprod - 1).to(device)

        # required for q_posterior
        self.posterior_mean_coef1 = self.betas * torch.sqrt(self.alphas_cumprod_prev) / (1. - self.alphas_cumprod).to(
            device)
        self.posterior_mean_coef2 = ((1. - self.alphas_cumprod_prev) * torch.sqrt(self.alphas) / (
                1. - self.alphas_cumprod)).to(device)

    def reconstruct_x0(self, x_t, t, noise):
        s1 = self.sqrt_inv_alphas_cumprod[t]
        s2 = self.sqrt_inv_alphas_cumprod_minus_one[t]
        s1 = s1.reshape(-1, 1)
        s2 = s2.reshape(-1, 1)
        return s1 * x_t - s2 * noise

    def q_posterior(self, x_0, x_t, t):
        s1 = self.posterior_mean_coef1[t]
        s2 = self.posterior_mean_coef2[t]
        s1 = s1.reshape(-1, 1)
        s2 = s2.reshape(-1, 1)
        mu = s1 * x_0 + s2 * x_t
        return mu

    def get_variance(self, t):
        if t == 0:
            return 0

        variance = self.betas[t] * (1. - self.alphas_cumprod_prev[t]) / (1. - self.alphas_cumprod[t])
        variance = variance.clip(1e-20)
        return variance

    def step(self, model_output, timestep, sample):
        t = timestep
        pred_original_sample = self.reconstruct_x0(sample, t, model_output)
        pred_prev_sample = self.q_posterior(pred_original_sample, sample, t)

        variance = 0
        if t > 0:
            noise = torch.randn_like(model_output)
            variance = (self.get_variance(t) ** 0.5) * noise

        pred_prev_sample = pred_prev_sample + variance

        return pred_prev_sample

    def add_noise(self, x_start, x_noise, timesteps):

        s1 = s1.reshape(-1, 1)
        s2 = s2.reshape(-1, 1)

        return s1 * x_start + s2 * x_noise

def smooth(x, window_len=10, window='hanning'):
    s = np.r_[x[window_len - 1:0:-1], x, x[-2:-window_len - 1:-1]]
    if window == 'flat':  # moving average
        w = np.ones(window_len, 'd')
    else:
        w = getattr(np, window)(window_len)
    y = np.convolve(w / w.sum(), s, mode='valid')
    return y

def smooth(x, window_len=10, window='hanning'):
    s = np.r_[x[window_len - 1:0:-1], x, x[-2:-window_len - 1:-1]]
    if window == 'flat':  # moving average
        w = np.ones(window_len, 'd')
    else:

    y = np.convolve(w / w.sum(), s, mode='valid')

% Introduce diffusion models and their recent success
Diffusion models have emerged as a powerful class of generative models, achieving state-of-the-art results in various domains such as image generation, audio synthesis, and molecular design \cite{yang2023diffusion}. % Explain the core concept of diffusion models
The core idea behind diffusion models, as introduced by \citet{sohl-dickstein15}, is to define a forward diffusion process that gradually adds Gaussian noise to the data, and then to learn a reverse diffusion process that removes this noise step by step. This reverse process is typically parameterized by a neural network and trained to predict the noise at each step, allowing for high-quality sample generation.

% Discuss the DDPM framework
The Denoising Diffusion Probabilistic Models (DDPM) framework, proposed by \citet{ddpm}, has become a cornerstone in the field of diffusion models. DDPMs define a fixed Markov chain for the forward process and use a neural network to learn the reverse process. This approach has shown remarkable success in generating high-quality samples while maintaining stable training dynamics.

% Introduce the concept of guided diffusion
Recent work has explored the concept of guided diffusion, where the generation process is steered towards desired outcomes. This guidance can take various forms, such as class-conditional generation or attribute-based control. However, most existing approaches focus on high-dimensional data like images, leaving the potential of guided diffusion for low-dimensional data largely unexplored.

% Description of the training process

We train for 10,000 steps with a batch size of 256. To ensure stability, we use gradient clipping with a maximum norm of 0.5.

% Description of the guidance implementation
To implement our guided diffusion approach, we modify the denoising process to incorporate the predefined path. At each denoising step $t$, we apply a guidance term that pulls the sample $x_t$ towards the target point $P(1-t/T)$ on the path, where $T$ is the total number of steps:

\begin{equation}
    x_{t-1} = \mu_\theta(x_t, t) + \lambda(P(1-t/T) - x_t) + \sigma_t z
\end{equation}

Here, $\mu_\theta$ is the denoising model, $\lambda$ is the guidance strength, $\sigma_t$ is the noise scale, and $z \sim \mathcal{N}(0, I)$.

% Description of the evaluation metrics

\begin{enumerate}

    \item Path adherence: We measure this as the mean squared error (MSE) between the generated samples and the nearest points on the guidance path. Lower MSE indicates better adherence to the desired trajectory.
\end{enumerate}

Recent advancements in diffusion models have led to various approaches for guided generation, primarily focusing on high-dimensional data such as images. \citet{ddpm} introduced the Denoising Diffusion Probabilistic Models (DDPM) framework, which forms the basis for many subsequent guided diffusion methods. This method has shown impressive results in conditional image generation tasks. Another significant development is classifier-free guidance, which eliminates the need for a separate classifier by directly conditioning the diffusion model on class information.

While these methods have demonstrated remarkable success in high-dimensional domains, they often rely on complex architectures and large-scale training, which may not be directly applicable or necessary for low-dimensional data.

\subsection{Controlled Generation in Other Generative Models}

While these methods offer valuable insights into controlled generation, they often face challenges such as mode collapse (in GANs) or blurry outputs (in VAEs) that diffusion models aim to address.

\subsection{Low-Dimensional Diffusion Models}

Low-dimensional diffusion models offer several advantages, including reduced computational requirements and easier visualization of the generation process. We modify the standard diffusion process to incorporate guidance. Given a predefined path $P: [0, 1] \rightarrow \mathbb{R}^d$ and a guidance strength parameter $\lambda \in [0, 1]$, we define the guided denoising step as:

\begin{equation}
