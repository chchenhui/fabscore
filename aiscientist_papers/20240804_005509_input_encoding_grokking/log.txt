 class Transformer(torch.nn.Module):                                            
     def __init__(                                                              
         self,                                                                  
         num_layers: int,                                                       
         dim_model: int,                                                        
         num_heads: int,                                                        
         vocab_size: int,                                                       
         output_size: int,                                                      
         seq_len: int,                                                          
         encoding_type: str,                                                    
     ):                                                                         
         super().__init__()                                                     
                                                                                
         self.encoding_type = encoding_type                                     
         if encoding_type == 'default':                                         
             self.token_embeddings = nn.Embedding(vocab_size, dim_model)        
         elif encoding_type == 'one-hot':                                       
             self.token_embeddings = nn.Linear(vocab_size, dim_model)           
         self.position_embeddings = nn.Embedding(seq_len, dim_model)            
         self.model = nn.Sequential(                                            
             *[DecoderBlock(dim_model, num_heads) for _ in range(num_layers)],  
             nn.LayerNorm(dim_model),                                           
             nn.Linear(dim_model, output_size),                                 
         )                                                                      
                                                                                
     def forward(self, inputs: Tensor):                                         
         if self.encoding_type == 'default':                                    
             batch_size, context_len = inputs.shape                             
             token_embedding = self.token_embeddings(inputs)                    
         elif self.encoding_type == 'one-hot':                                  
             batch_size, context_len, _ = inputs.shape                          
             token_embedding = self.token_embeddings(inputs.float())  # Convert 
 to float                                                                       
         else:                                                                  
             raise ValueError(f"Unknown encoding type: {self.encoding_type}")   
                                                                                
         positions = repeat(                                                    
             torch.arange(context_len, device=inputs.device), "p -> b p",       
 b=batch_size                                                                   
         )                                                                      
         position_embedding = self.position_embeddings(positions)               
                                                                                
         embedding = token_embedding + position_embedding                       
                                                                                
         embedding = rearrange(embedding, "b s d -> s b d")                     
                                                                                
         return self.model(embedding)
