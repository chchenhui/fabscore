def compute_pairwise_distances(batch):
    """Compute pairwise distances within a batch."""
    diff = batch.unsqueeze(1) - batch.unsqueeze(0)
    distances = torch.norm(diff, dim=-1)
    return distances

def manifold_preserving_loss(noise_pred, noise, noisy, original, timesteps, noise_scheduler):
    """Compute the manifold-preserving loss."""
    mse_loss = F.mse_loss(noise_pred, noise)
    
    # Reconstruct the denoised samples
    denoised = noise_scheduler.reconstruct_x0(noisy, timesteps, noise_pred)
    
    # Compute pairwise distances for original and denoised samples
    original_distances = compute_pairwise_distances(original)
    denoised_distances = compute_pairwise_distances(denoised)
    
    # Compute the distance preservation loss
    distance_loss = F.mse_loss(denoised_distances, original_distances)
    
    # Combine the losses
    total_loss = mse_loss + 0.1 * distance_loss  # Adjust the weight as needed
    
    return total_loss, mse_loss, distance_loss

class AdaptiveWeightScheduler:
    def __init__(self, initial_weight, patience=100, factor=0.9, min_weight=0.01, max_weight=0.5):
        self.weight = initial_weight
        self.best_loss = float('inf')
        self.patience = patience
        self.factor = factor
        self.min_weight = min_weight
        self.max_weight = max_weight
        self.counter = 0

    def step(self, loss):
        if loss < self.best_loss:
            self.best_loss = loss
            self.counter = 0
        else:
            self.counter += 1
            if self.counter >= self.patience:
                self.weight = max(self.min_weight, min(self.max_weight, self.weight * self.factor))
                self.counter = 0
        return self.weight

% Description of datasets
Our experiments were conducted on four low-dimensional datasets: circle, dino, line, and moons. These datasets were chosen to represent a diverse range of low-dimensional structures commonly encountered in data analysis and visualization tasks. Each dataset consists of 100,000 samples in 2D space, generated using the datasets module from scikit-learn \cite{scikit-learn}. Specifically:
\begin{itemize}
    \item Circle: Points uniformly distributed on a circle with added Gaussian noise.
    \item Dino: A custom dataset resembling the silhouette of a dinosaur.
    \item Line: Points uniformly distributed along a diagonal line with added Gaussian noise.
    \item Moons: Two interleaving half circles, forming a shape similar to two crescent moons.
\end{itemize}

% Model architecture and training details
We implemented our manifold-preserving diffusion model using a multi-layer perceptron (MLP) architecture. The model consists of an embedding layer for time steps (128 dimensions), followed by three residual blocks (256 hidden units each), and a final output layer (2 dimensions). We used sinusoidal embeddings for both time steps and input data, which helps capture high-frequency patterns in low-dimensional data \cite{ddpm}. ReLU activation functions were used throughout the network.

The diffusion process was implemented with 100 timesteps, using a linear beta schedule as described in \citet{ddpm}. We trained the models for 10,000 steps using the AdamW optimizer \cite{adamw} with a learning rate of 3e-4 and a batch size of 256. To stabilize training and improve sample quality, we employed an exponential moving average (EMA) of the model parameters with a decay rate of 0.995, updating every 10 steps.

\begin{equation}
\mathcal{L} = \mathcal{L}_\text{MSE} + \lambda \mathcal{L}_\text{distance}
\end{equation}

where $\mathcal{L}_\text{MSE}$ is the standard DDPM loss, $\mathcal{L}_\text{distance}$ is the mean squared error between pairwise distances of original and reconstructed samples, and $\lambda$ is a weighting factor. The pairwise distances are computed efficiently within mini-batches using vectorized operations, ensuring computational feasibility even for large batch sizes.

\subsection{Manifold-Preserving Loss Function}

The core of our approach is a modified loss function that combines the standard diffusion model objective with a term that preserves pairwise distances between data points. Let $\mathcal{L}_\text{DDPM}$ be the standard denoising diffusion probabilistic models (DDPM) loss \cite{ddpm}. Our manifold-preserving loss $\mathcal{L}_\text{MP}$ is defined as:

\begin{equation}
\mathcal{L}_\text{MP} = \mathcal{L}_\text{DDPM} + \lambda \mathcal{L}_\text{distance}
\end{equation}

where $\lambda$ is a weighting factor and $\mathcal{L}_\text{distance}$ is the distance preservation term. We define $\mathcal{L}_\text{distance}$ as:

\begin{equation}
\mathcal{L}_\text{distance} = \mathbb{E}_{\mathbf{x}_0, t, \epsilon} \left[ \frac{1}{N^2} \sum_{i,j} \left( \| \hat{\mathbf{x}}_{0,i} - \hat{\mathbf{x}}_{0,j} \|_2 - \| \mathbf{x}_{0,i} - \mathbf{x}_{0,j} \|_2 \right)^2 \right]
\end{equation}

where $\mathbf{x}_0$ is the original data, $\hat{\mathbf{x}}_0$ is the reconstructed data, $N$ is the batch size, and $\| \cdot \|_2$ denotes the Euclidean norm.

This loss term encourages the model to maintain pairwise distances between points in the reconstructed data, thus preserving the local structure of the manifold. The expectation is taken over the data distribution, timesteps, and noise samples.
