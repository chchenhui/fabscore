     1	# FIR-CUSUM-epsilon sequential test controller (proposed method).
     2	# Accumulates evidence of sustained upward drift via one-sided CUSUM statistic:
     3	#   S_t = max(0, S_{t-1} + r_t - k), rollback when S_t > h.
     4	# r_t = (nu_t - mu_0) / (sigma_0 + 1e-8) is the standardized innovation.
     5	# Uses Fast Initial Response (FIR): on rollback S_t resets to h*reset_fraction
     6	# instead of 0, so the CUSUM re-triggers quickly during sustained perturbation.
     7	# On accept: update y_hat via EMA, save new snapshot.
     8	
     9	from cusum_controller.controllers.base_controller import BaseController
    10	
    11	
    12	class CUSUMController(BaseController):
    13	    def __init__(self, alpha=0.1, k=0.5, h=None, mu_0=0.0, sigma_0=1.0,
    14	                 reset_fraction=0.5):
    15	        if h is None:
    16	            raise ValueError("h must be provided (run calibration first)")
    17	        self.alpha = alpha
    18	        self.k = k
    19	        self.h = h
    20	        self.mu_0 = mu_0
    21	        self.sigma_0 = sigma_0
    22	        self.reset_fraction = reset_fraction
    23	        self.y_hat = None
    24	        self.S_t = 0.0
    25	
    26	    def initialize(self, y0):
    27	        self.y_hat = y0
    28	        self.S_t = 0.0
    29	
    30	    def decide(self, y_proposed):
    31	        nu = y_proposed - self.y_hat
    32	        r_t = (nu - self.mu_0) / (self.sigma_0 + 1e-8)
    33	        self.S_t = max(0.0, self.S_t + r_t - self.k)
    34	
    35	        if self.S_t > self.h:
    36	            accepted = False
    37	            self.S_t = self.h * self.reset_fraction
    38	        else:
    39	            accepted = True
    40	            self.y_hat = (1.0 - self.alpha) * self.y_hat + self.alpha * y_proposed
    41	
    42	        info = {
    43	            "y_hat": self.y_hat,
    44	            "innovation": nu,
    45	            "r_t": r_t,
    46	            "S_t": self.S_t,
    47	            "accepted": accepted,
    48	        }
    49	
    50	        return accepted, info
