   168	    def _calculate_metrics(self, model: BaseAlignmentModel, raw_data: Dict) -> Dict[str, float]:
   169	        """
   170	        Calculate evaluation metrics for a model based on raw simulation data.
   171	        
   172	        Args:
   173	            model: Alignment model to evaluate
   174	            raw_data: Raw simulation data
   175	            
   176	        Returns:
   177	            Dictionary with evaluation metrics
   178	        """
   179	        metrics = {}
   180	        
   181	        # Calculate average alignment score
   182	        alignment_scores = raw_data["alignment_scores"][model.name]
   183	        metrics["avg_alignment_score"] = np.mean(alignment_scores)
   184	        
   185	        # Calculate final alignment score
   186	        metrics["final_alignment_score"] = alignment_scores[-1]
   187	        
   188	        # Calculate adaptation accuracy
   189	        # This is averaged across all time steps and measures how well the model tracks human values
   190	        human_values_array = np.array(raw_data["human_values"])[:, 0, :]  # First agent only
   191	        model_values_array = np.array(raw_data["model_values"][model.name])
   192	        
   193	        # Calculate Euclidean distance at each time step
   194	        distances = np.linalg.norm(human_values_array - model_values_array, axis=1)
   195	        avg_distance = np.mean(distances)
   196	        
   197	        # Convert to adaptation accuracy (1 - normalized distance)
   198	        # Maximum distance in n-dimensional space with values summing to 1 is sqrt(2)
   199	        max_distance = np.sqrt(2)
   200	        metrics["adaptation_accuracy"] = 1 - (avg_distance / max_distance)
