   327	    def update(self, human_values: ValueSystem) -> None:
   328	        """
   329	        Update the model's value system based on human values.
   330	        
   331	        Args:
   332	            human_values: Human value system to align with
   333	        """
   334	        # Update each value dimension with level-specific adaptation rates
   335	        for dim in self.value_dimensions:
   336	            current_value = self.value_system.get_value(dim)
   337	            human_value = human_values.get_value(dim)
   338	            adaptation_rate = self._get_adaptation_rate(dim)
   339	            
   340	            # Higher confidence leads to faster adaptation
   341	            effective_rate = adaptation_rate * self.value_confidence[dim]
   342	            
   343	            # Linear interpolation between current and target values
   344	            new_value = (1 - effective_rate) * current_value + effective_rate * human_value
   345	            self.value_system.set_value(dim, new_value)
   346	            
   347	            # Update confidence based on consistency of observations
   348	            # If human values are stable, confidence increases; if they change a lot, confidence decreases
   349	            value_diff = abs(human_value - current_value)
   350	            conf_update = 0.01 * (1 - value_diff)  # Small updates to confidence
   351	            self.value_confidence[dim] = max(0.1, min(1.0, self.value_confidence[dim] + conf_update))
   352	            
   353	        # Record current values in history
   354	        self.value_history.append(self.value_system.values.copy())
   355	        
   356	        # Calculate and record alignment score
   357	        alignment_score = self.calculate_alignment_score(human_values)
   358	        self.alignment_scores.append(alignment_score)
